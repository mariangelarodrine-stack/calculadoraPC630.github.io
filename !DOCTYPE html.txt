/* =====================================================
   FORMATO NÚMEROS
===================================================== */

function formatNumber(number){

    return number.toLocaleString("es-PE");

}


/* =====================================================
   CRECIMIENTO
===================================================== */

function calculateGrowth(ly,current){

    if(ly === 0){

        return null;

    }

    return ((current / ly) - 1) * 100;

}


function formatGrowth(value){

    if(value === null){

        return "N/D";

    }

    return (value >= 0 ? "+" : "") +
           value.toFixed(1) +
           "%";

}


/* =====================================================
   DIBUJAR GRÁFICO
===================================================== */

function drawChart(animate=true){

    const client =
        clients[Number(select.value)];


    const values25 =
        client.values2025;

    const values26 =
        client.values2026;


    /* DIMENSIONES */

    const W = 1200;

    const H = 480;

    const left = 65;

    const right = 30;

    const top = 30;

    const bottom = 75;


    const chartWidth =
        W - left - right;

    const chartHeight =
        H - top - bottom;


    /* ESCALA */

    const maximum =
        Math.max(
            ...values25,
            ...values26
        );


    const step =
        Math.ceil(
            maximum / 4 / 5000
        ) * 5000;


    const maxY =
        step * 4;


    /* POSICIONES X */

    const xPositions = [

        left,

        left + chartWidth / 2,

        W - right

    ];


    /* POSICIÓN Y */

    function y(value){

        return top +
            ((maxY-value) / maxY)
            * chartHeight;

    }


    let html = "";


    /* =================================================
       GRID
    ================================================= */

    for(let i=0;i<=4;i++){

        const value =
            maxY -
            (maxY/4)*i;

        const yy =
            y(value);


        html += `

            <line
                class="grid"
                x1="${left}"
                y1="${yy}"
                x2="${W-right}"
                y2="${yy}">
            </line>

            <text
                class="axis-label"
                x="${left-10}"
                y="${yy+4}"
                text-anchor="end">

                ${formatNumber(Math.round(value))}

            </text>

        `;

    }


    /* =================================================
       PUNTOS
    ================================================= */

    const points25 =
        values25.map(
            (value,i)=>[
                xPositions[i],
                y(value)
            ]
        );


    const points26 =
        values26.map(
            (value,i)=>[
                xPositions[i],
                y(value)
            ]
        );


    function createPath(points){

        return points
            .map(
                (point,index)=>
                (index===0 ? "M" : "L") +
                point[0] +
                " " +
                point[1]
            )
            .join(" ");

    }


    /* =================================================
       LÍNEAS
    ================================================= */

    html += `

        <path
            id="line25"
            class="line-2025"
            d="${createPath(points25)}">
        </path>


        <path
            id="line26"
            class="line-2026"
            d="${createPath(points26)}">
        </path>

    `;


    /* =================================================
       MESES Y PUNTOS
    ================================================= */

    months.forEach((month,index)=>{

        const x =
            xPositions[index];


        const y25 =
            y(values25[index]);


        const y26 =
            y(values26[index]);


        /* MES */

        html += `

            <text
                class="month-label"
                x="${x}"
                y="${H-bottom+28}"
                text-anchor="middle">

                ${month}

            </text>

        `;


        /* PUNTO 2025 */

        html += `

            <circle
                class="point-2025"
                cx="${x}"
                cy="${y25}"
                r="5"
                data-year="2025"
                data-month="${index}">
            </circle>

        `;


        /* PUNTO 2026 */

        html += `

            <circle
                class="point-2026"
                cx="${x}"
                cy="${y26}"
                r="5"
                data-year="2026"
                data-month="${index}">
            </circle>

        `;


        /* VALOR 2025 */

        html += `

            <text
                class="value-2025"
                x="${x}"
                y="${y25-12}"
                text-anchor="middle">

                ${formatNumber(values25[index])}

            </text>

        `;


        /* VALOR 2026 */

        html += `

            <text
                class="value-2026"
                x="${x}"
                y="${y26+22}"
                text-anchor="middle">

                ${formatNumber(values26[index])}

            </text>

        `;

    });


    /* INSERTAR SVG */

    svg.innerHTML = html;


    /* =================================================
       ANIMACIÓN
    ================================================= */

    ["line25","line26"].forEach(id=>{

        const path =
            document.getElementById(id);

        const length =
            path.getTotalLength();


        path.style.strokeDasharray =
            length;

        path.style.strokeDashoffset =
            animate ? length : 0;


        if(animate){

            requestAnimationFrame(()=>{

                path.style.transition =
                    "stroke-dashoffset 0.7s ease";

                path.style.strokeDashoffset =
                    "0";

            });

        }

    });


    /* =================================================
       TOOLTIP
    ================================================= */

    svg.querySelectorAll("circle")
        .forEach(point=>{

        point.addEventListener(
            "mouseenter",
            function(){

                const monthIndex =
                    Number(this.dataset.month);

                const year =
                    this.dataset.year;


                const value =
                    year === "2025"
                    ? values25[monthIndex]
                    : values26[monthIndex];


                const ly =
                    year === "2025"
                    ? values25[monthIndex]
                    : values25[monthIndex];


                const actual =
                    values26[monthIndex];


                const growth =
                    calculateGrowth(
                        ly,
                        actual
                    );


                tooltip.innerHTML = `

                    <strong>
                        ${client.name}
                    </strong>

                    <br>

                    ${months[monthIndex]}
                    ${year}:
                    ${formatNumber(value)}
                    cajas eq.

                    <br>

                    Crecimiento vs LY:
                    <strong>
                        ${formatGrowth(growth)}
                    </strong>

                `;


                tooltip.style.display =
                    "block";

            }
        );


        point.addEventListener(
            "mousemove",
            function(event){

                tooltip.style.left =
                    event.clientX + 15 + "px";

                tooltip.style.top =
                    event.clientY + 15 + "px";

            }
        );


        point.addEventListener(
            "mouseleave",
            function(){

                tooltip.style.display =
                    "none";

            }
        );

    });


    /* =================================================
       TÍTULO
    ================================================= */

    document.getElementById(
        "clientName"
    ).textContent =
        client.name;


    /* =================================================
       CRECIMIENTOS
    ================================================= */

    const growths =
        values25.map(
            (value,index)=>
            calculateGrowth(
                value,
                values26[index]
            )
        );


    const growthElements = [

        document.getElementById("growthJun"),

        document.getElementById("growthJul"),

        document.getElementById("growthAgo")

    ];


    growthElements.forEach(
        (element,index)=>{

        const value =
            growths[index];


        element.textContent =
            formatGrowth(value);


        element.className =
            value !== null &&
            value >= 0
            ? "positive"
            : "negative";

    });

}
